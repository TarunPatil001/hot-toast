💻 Code Examples
Basic Usage
toast('Hello World!');
toast.success('Success!');
toast.error('Error!');
toast.loading('Loading...');
Custom Options
toast('Good Job!', {
    icon: '👏',
    duration: 5000,
    style: 'background: #333; color: #fff;'
});
Promise Handling
const myPromise = fetch('/api/data');

toast.promise(myPromise, {
    loading: 'Loading...',
    success: 'Data loaded!',
    error: 'Failed to load!'
});
Configuration
toast.config({
    position: 'top-center',
    duration: 4000,
    maxToasts: 5,
    reverseOrder: false
});
Dismiss Toasts
const id = toast('Hello');
toast.dismiss(id);  // Dismiss specific
toast.dismiss();    // Dismiss all
